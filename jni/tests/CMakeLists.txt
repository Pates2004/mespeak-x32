cmake_minimum_required(VERSION 3.22.1)
project(mespeak-native-cancellation-tests LANGUAGES C CXX)

if(NOT ANDROID)
    message(FATAL_ERROR "Use the Android NDK toolchain for these bounded fixtures.")
endif()
set(SONIC_SOURCE_DIR "" CACHE PATH "Existing checkout of the product's pinned Sonic revision")
if(NOT EXISTS "${SONIC_SOURCE_DIR}/sonic.c" OR NOT EXISTS "${SONIC_SOURCE_DIR}/sonic.h")
    message(FATAL_ERROR "Set SONIC_SOURCE_DIR to the product's pinned Sonic source checkout.")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(cancellation_delivery_test cancellation_delivery_test.c "${SONIC_SOURCE_DIR}/sonic.c")
add_executable(wavegen_stop_test wavegen_stop_test.cpp)
foreach(fixture cancellation_delivery_test wavegen_stop_test)
    target_include_directories(${fixture} PRIVATE ../../engine/src ../include "${SONIC_SOURCE_DIR}")
    target_compile_definitions(${fixture} PRIVATE USE_MBROLA=0 NDEBUG)
    target_compile_options(${fixture} PRIVATE -fno-exceptions -ffunction-sections -fdata-sections)
    target_link_options(${fixture} PRIVATE -Wl,--gc-sections -Wl,-z,max-page-size=16384)
    target_link_libraries(${fixture} PRIVATE log m)
endforeach()
